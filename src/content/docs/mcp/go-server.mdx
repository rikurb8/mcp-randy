### Basic Server Structure

```go
package main

import (
    "context"
    "fmt"
    "log"
    "os"
    
    // When mcp-go is stable:
    // "github.com/mark3labs/mcp-go/mcp"
    // "github.com/mark3labs/mcp-go/server"
)

// Server represents our MCP server
type Server struct {
    name    string
    version string
}

// NewServer creates a new MCP server instance
func NewServer(name, version string) *Server {
    return &Server{
        name:    name,
        version: version,
    }
}

// Tool represents an MCP tool
type Tool struct {
    Name        string                 `json:"name"`
    Description string                 `json:"description"`
    Parameters  map[string]interface{} `json:"parameters"`
}

// ToolResult represents the result of a tool execution
type ToolResult struct {
    Success bool        `json:"success"`
    Result  interface{} `json:"result,omitempty"`
    Error   string      `json:"error,omitempty"`
}

func main() {
    server := NewServer("randy-demo-server", "1.0.0")
    
    // Register tools
    server.RegisterCalculatorTool()
    server.RegisterGreetingTool()
    server.RegisterFileListTool()
    
    // Start server (stdio transport)
    log.Printf("Starting MCP server: %s v%s", server.name, server.version)
    server.Start()
}
```

### File Operations Tool

```go
// RegisterFileListTool adds file listing capability
func (s *Server) RegisterFileListTool() {
    tool := Tool{
        Name:        "list_files",
        Description: "List files in a directory",
        Parameters: map[string]interface{}{
            "type": "object",
            "properties": map[string]interface{}{
                "path": map[string]interface{}{
                    "type":        "string",
                    "description": "Directory path to list (default: current directory)",
                },
                "include_hidden": map[string]interface{}{
                    "type":        "boolean",
                    "description": "Include hidden files (default: false)",
                },
            },
        },
    }
    
    s.registerTool(tool, s.handleFileList)
}

// handleFileList processes file listing requests
func (s *Server) handleFileList(ctx context.Context, params map[string]interface{}) ToolResult {
    path, _ := params["path"].(string)
    if path == "" {
        path = "."
    }
    
    includeHidden, _ := params["include_hidden"].(bool)
    
    entries, err := os.ReadDir(path)
    if err != nil {
        return ToolResult{
            Success: false,
            Error:   fmt.Sprintf("failed to read directory: %v", err),
        }
    }
    
    var files []map[string]interface{}
    for _, entry := range entries {
        name := entry.Name()
        
        // Skip hidden files unless requested
        if !includeHidden && name[0] == '.' {
            continue
        }
        
        info, err := entry.Info()
        if err != nil {
            continue
        }
        
        files = append(files, map[string]interface{}{
            "name":      name,
            "size":      info.Size(),
            "is_dir":    entry.IsDir(),
            "modified":  info.ModTime().Unix(),
        })
    }
    
    return ToolResult{
        Success: true,
        Result: map[string]interface{}{
            "path":  path,
            "files": files,
            "count": len(files),
        },
    }
}
```

### Error Handling

```go
// MCPError represents an MCP protocol error
type MCPError struct {
    Code    int    `json:"code"`
    Message string `json:"message"`
    Data    interface{} `json:"data,omitempty"`
}

func (e MCPError) Error() string {
    return fmt.Sprintf("MCP error %d: %s", e.Code, e.Message)
}

// Common MCP error codes
const (
    ErrorInvalidRequest     = -32600
    ErrorMethodNotFound     = -32601  
    ErrorInvalidParams      = -32602
    ErrorInternalError      = -32603
    ErrorParseError         = -32700
)

// validateToolParams validates tool parameters
func validateToolParams(tool Tool, params map[string]interface{}) error {
    schema := tool.Parameters
    required, _ := schema["required"].([]string)
    
    for _, field := range required {
        if _, exists := params[field]; !exists {
            return MCPError{
                Code:    ErrorInvalidParams,
                Message: fmt.Sprintf("missing required parameter: %s", field),
            }
        }
    }
    
    return nil
}
```

### Protocol Implementation

```go
// MCPRequest represents an incoming MCP request
type MCPRequest struct {
    JSONRPC string      `json:"jsonrpc"`
    Method  string      `json:"method"`
    Params  interface{} `json:"params,omitempty"`
    ID      interface{} `json:"id,omitempty"`
}

// MCPResponse represents an MCP response
type MCPResponse struct {
    JSONRPC string      `json:"jsonrpc"`
    Result  interface{} `json:"result,omitempty"`
    Error   *MCPError   `json:"error,omitempty"`
    ID      interface{} `json:"id,omitempty"`
}

// handleMCPRequest processes incoming MCP requests
func (s *Server) handleMCPRequest(request MCPRequest) MCPResponse {
    response := MCPResponse{
        JSONRPC: "2.0",
        ID:      request.ID,
    }
    
    switch request.Method {
    case "initialize":
        response.Result = s.handleInitialize(request.Params)
    case "tools/list":
        response.Result = s.handleToolsList()
    case "tools/call":
        result, err := s.handleToolCall(request.Params)
        if err != nil {
            response.Error = &MCPError{
                Code:    ErrorInternalError,
                Message: err.Error(),
            }
        } else {
            response.Result = result
        }
    default:
        response.Error = &MCPError{
            Code:    ErrorMethodNotFound,
            Message: fmt.Sprintf("method not found: %s", request.Method),
        }
    }
    
    return response
}
```

```go
import (
    "context"

func TestCalculatorTool(t *testing.T) {
    server := NewServer("test", "1.0.0")
    server.RegisterCalculatorTool()
    
    tests := []struct {
        name       string
        params     map[string]interface{}
        wantResult bool
    }{
        {
            name:       "simple addition",
            params:     map[string]interface{}{"expression": "2 + 2"},
            wantResult: true,
        },
        {
            name:       "missing expression",
            params:     map[string]interface{}{},
            wantResult: false,
        },
    }
    
    for _, tt := range tests {
        t.Run(tt.name, func(t *testing.T) {
            result := server.handleCalculate(context.Background(), tt.params)
            if result.Success != tt.wantResult {
                t.Errorf("expected success=%v, got=%v", tt.wantResult, result.Success)
            }
        })
    }
}
```
